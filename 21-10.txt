+## 5. Decisiones y puntos que debes conocer
+
+1. **Supabase no se integró.** El prompt lo lista como conector, pero choca con el requisito de MSSQL; usar ambos duplicaría la fuente de verdad. Si quieres Supabase solo para *storage de respaldo*, es una adición aparte.
+2. **Conectores (Calendar, Gmail, Jira/Confluence, Vercel) y skills de proceso (`to-tickets`, `triage`, `wayfinder`…) no se invocaron:** crean cosas en sistemas externos que no pediste explícitamente. Dímelo y los uso.
+3. **`graphify` es una skill de grafos de código**, no de gráficas. Los reportes visuales del panel (barras de cobrado por día) son SVG/CSS propios; `img2threejs` no aplica todavía: solo se dejó la ruta `/ir/admisiones`.
+4. **"Al día"** es un estado derivado: tras aprobar, la respuesta devuelve `estatus_pago` recalculado; es "Al día" solo si no quedan otras cuotas vencidas (si las hay, sigue "Moroso", que es lo correcto).
+5. **Matrícula sin precio inventado:** `MATRICULA_USD` nace en `0.00` y las inscripciones responden `503 CONFIG_INCOMPLETA` hasta que el admin la fije (`PUT /api/configuracion/matricula`). Igual las cuentas Zelle/Binance/PayPal/Pago Móvil: se cargan por `POST /api/configuracion/cuentas`.
+6. **Drive:** una cuenta de servicio no tiene cuota propia. La carpeta padre debe estar en una **Unidad Compartida** o las subidas fallarán con `storageQuotaExceeded`.
+7. **Transacción abierta durante la subida a Drive** (para poder revertir). Mitigado con un tope de 4 inscripciones simultáneas (`ADMISIONES_CONCURRENCIA_MAX`) → `503 SERVIDOR_OCUPADO` + `Retry-After`. Con volumen alto conviene pasar a subida asíncrona con cola.
+8. **Limitadores en memoria:** válidos para una sola instancia; con réplicas, moverlos a Redis/WAF.
+9. **API BCV:** por defecto `https://ve.dolarapi.com/v1/dolares/oficial` (campo `promedio`), configurable con `BCV_API_URL`. Es un agregador, no el BCV directamente: confirma que la fuente sea aceptable para cobranza.
+10. Las alertas por WhatsApp fuera de la ventana de 24 h de Meta pueden exigir una **plantilla aprobada**; hoy se envía texto libre. Si Meta rechaza, queda `FALLIDA` y la API responde 502.

+'use strict';
+// Pool único de MSSQL (node-mssql / tedious). Perezoso y compartido: la primera consulta crea y verifica el pool;
+// si la conexión inicial falla se descarta la promesa para que el siguiente intento reconecte (no queda "envenenado").
+const sql = require('mssql');
+const cfg = require('./env');
+
+let poolPromise = null;
+
+function crearConfig() {
+  const d = cfg.db;
+  if (!d.server || !d.database || !d.user || !d.password) {
+    throw new Error('[db] Faltan DB_SERVER, DB_NAME, DB_USER o DB_PASSWORD en el entorno.');
+  }
+  return {
+    server: d.server,
+    port: d.port,
+    database: d.database,
+    user: d.user,
+    password: d.password,
+    connectionTimeout: d.connectTimeoutMs,
+    requestTimeout: d.requestTimeoutMs,
+    options: { encrypt: d.encrypt, trustServerCertificate: d.trustServerCertificate, enableArithAbort: true },
+    pool: { max: d.poolMax, min: d.poolMin, idleTimeoutMillis: d.idleMs },
+  };
+}
+
+function getPool() {
+  if (!poolPromise) {
+    let pool;
+    try {
+      pool = new sql.ConnectionPool(crearConfig());
+    } catch (err) {
+      return Promise.reject(err);
+    }
+    // Sin este listener, una conexión inactiva que se cae tumba el proceso.
+    pool.on('error', (err) => console.error('[db] Error en el pool:', err.message));
+    poolPromise = pool.connect().catch((err) => {
+      poolPromise = null;
+      console.error('[db] Falló la conexión inicial:', err.message);
+      throw err;
+    });
+  }
+  return poolPromise;
+}
+
+async function closePool() {
+  if (!poolPromise) return;
+  const pendiente = poolPromise;
+  poolPromise = null;
+  const pool = await pendiente.catch(() => null);
+  if (pool) await pool.close();
+}
+
+/**
+ * Ejecuta `fn(tx)` dentro de una transacción. COMMIT si resuelve, ROLLBACK si lanza (el error original se relanza).
+ * Una transacción por petición: el cliente se devuelve al pool siempre (sin fugas de conexiones).
+ */
+async function conTransaccion(fn, nivel = sql.ISOLATION_LEVEL.READ_COMMITTED) {
+  const pool = await getPool();
+  const tx = new sql.Transaction(pool);
+  await tx.begin(nivel);
+  try {
+    const resultado = await fn(tx);
+    await tx.commit();
+    return resultado;
+  } catch (err) {
+    try {
+      await tx.rollback();
+    } catch (rollbackErr) {
+      console.error('[db] Rollback falló:', rollbackErr.message);
+    }
+    throw err;
+  }
+}
+
+/** Atajo para consultas parametrizadas. `params` = { nombre: [tipo, valor] }. Nunca concatenar SQL. */
+async function ejecutar(textoSql, params = {}, ejecutor) {
+  const request = (ejecutor ? new sql.Request(ejecutor) : (await getPool()).request());
+  for (const [nombre, [tipo, valor]] of Object.entries(params)) request.input(nombre, tipo, valor);
+  return request.query(textoSql);
+}

+-- 002_vistas.sql — Clasificación en tiempo real de estudiantes (Activo / Fuera de la Academia / Moroso).
+-- "Hoy" = día calendario de Caracas (UTC-4 fijo). Una cuota Pendiente o En Revisión con vencimiento
+-- anterior a hoy convierte al estudiante en Moroso hasta que un administrador apruebe la referencia.
+
+CREATE OR ALTER VIEW dbo.vw_estudiantes_estado AS
+SELECT
+  e.id,
+  e.representante_id,
+  e.nombres,
+  e.apellidos,
+  e.grado_solicitado,
+  e.estatus_academico,
+  CASE
+    WHEN e.estatus_academico = N'Fuera de la Academia'                      THEN N'Fuera de la Academia'
+    WHEN e.estatus_academico = N'Pendiente por Validación Administrativa'   THEN N'Pendiente por Validación Administrativa'
+    WHEN ISNULL(m.cuotas_vencidas, 0) > 0                                   THEN N'Moroso'
+    ELSE N'Activo'
+  END AS estado_dinamico,
+  CASE
+    WHEN e.estatus_academico = N'Fuera de la Academia'                      THEN N'N/A'
+    WHEN e.estatus_academico = N'Pendiente por Validación Administrativa'   THEN N'Pendiente de Validación'
+    WHEN ISNULL(m.cuotas_vencidas, 0) > 0                                   THEN N'Moroso'
+    ELSE N'Al día'
+  END AS estatus_pago,
+  ISNULL(m.cuotas_vencidas, 0)       AS cuotas_vencidas,
+  ISNULL(m.deuda_vencida_usd, 0)     AS deuda_vencida_usd,
+  m.proximo_vencimiento
+FROM dbo.estudiantes AS e
+CROSS APPLY (SELECT CAST(SWITCHOFFSET(SYSDATETIMEOFFSET(), N'-04:00') AS DATE) AS hoy) AS h
+OUTER APPLY (
+  SELECT
+    SUM(CASE WHEN c.fecha_vencimiento < h.hoy THEN 1 ELSE 0 END)              AS cuotas_vencidas,
+    SUM(CASE WHEN c.fecha_vencimiento < h.hoy THEN c.monto_usd ELSE 0 END)    AS deuda_vencida_usd,
+    MIN(c.fecha_vencimiento)                                                  AS proximo_vencimiento
+  FROM dbo.cuotas AS c
+  WHERE c.estudiante_id = e.id
+    AND c.estado IN (N'Pendiente', N'En Revisión')
+) AS m;
+GO

+'use strict';
+const jwt = require('jsonwebtoken');
+const cfg = require('../config/env');
+const { e } = require('../errors/AppError');
+
+function firmarToken(usuario) {
+  return jwt.sign({ sub: usuario.id, rol: usuario.rol, nombre: usuario.nombre }, cfg.jwt.secret, {
+    algorithm: 'HS256',
+    expiresIn: cfg.jwt.expiraEn,
+  });
+}
+
+/** Exige `Authorization: Bearer <jwt>`. Fija req.usuario = { id, rol, nombre }. */
+function requireAuth(req, _res, next) {
+  const cabecera = req.get('authorization') || '';
+  const m = /^Bearer\s+(.+)$/i.exec(cabecera);
+  if (!m) return next(e.noAutenticado());
+  try {
+    const p = jwt.verify(m[1], cfg.jwt.secret, { algorithms: ['HS256'] });
+    req.usuario = { id: Number(p.sub), rol: p.rol, nombre: p.nombre };
+    return next();
+  } catch (err) {
+    return next(e.noAutenticado(err.name === 'TokenExpiredError' ? 'La sesión expiró. Inicia sesión de nuevo.' : 'Token inválido.'));
+  }
+}
+
+/** requireRol('admin') o requireRol('admin','secretaria'). */
+function requireRol(...roles) {
+  return (req, _res, next) => {
+    if (!req.usuario) return next(e.noAutenticado());
+    if (!roles.includes(req.usuario.rol)) return next(e.prohibido());
+    return next();
+  };
+}
+

+/** Traduce cualquier error a { status, codigo, mensaje, detalles?, headers? }. Nunca expone SQL ni stack al cliente. */
+function normalizar(err) {
+  if (err instanceof AppError) {
+    return { status: err.status, codigo: err.codigo, mensaje: err.message, detalles: err.detalles, headers: err.headers };
+  }
+  // Cuerpo JSON malformado o demasiado grande (body-parser).
+  if (err && err.type === 'entity.parse.failed') {
+    return { status: 400, codigo: 'JSON_INVALIDO', mensaje: 'El cuerpo de la petición no es JSON válido.' };
+  }

+'use strict';
+const { AppError } = require('../errors/AppError');
+
+/**
+ * Limitador en memoria por IP (ventana fija). Suficiente para una sola instancia;
+ * con varias réplicas hay que moverlo a Redis o a la capa de Vercel/WAF.
+ */
+function crearLimitador({ ventanaMs = 15 * 60_000, max = 10, codigo = 'DEMASIADAS_SOLICITUDES', mensaje = 'Demasiadas solicitudes. Espera unos minutos.', soloFallos = false } = {}) {
+  const cubetas = new Map();
+  const limpieza = setInterval(() => {
+    const ahora = Date.now();
+    for (const [k, v] of cubetas) if (v.reinicia <= ahora) cubetas.delete(k);
+  }, Math.min(ventanaMs, 60_000));
+  limpieza.unref();
+
+  return function limitador(req, res, next) {
+    const ahora = Date.now();
+    const clave = req.ip || 'desconocida';
+    let c = cubetas.get(clave);
+    if (!c || c.reinicia <= ahora) {
+      c = { n: 0, reinicia: ahora + ventanaMs };
+      cubetas.set(clave, c);
+    }
+    if (c.n >= max) {
+      const espera = Math.max(1, Math.ceil((c.reinicia - ahora) / 1000));
+      return next(new AppError(429, codigo, mensaje).conCabeceras({ 'Retry-After': String(espera) }));
+    }
+    if (soloFallos) {
+      res.on('finish', () => { if (res.statusCode >= 400) c.n += 1; });
+    } else {
+      c.n += 1;
+    }
+    next();
+  };
+}
+
+module.exports = { crearLimitador };

+    return r.recordset[0] || null;
+  },
+  async leerTodos() {
+    return (await ejecutar('SELECT codigo, nombre, descripcion, precio_usd, habilitado, habilitado_en FROM dbo.modulos_premium ORDER BY precio_usd')).recordset;

+/**
+ * Candado de módulos premium. El estado se lee de la base (cache de 15 s para no consultar en cada petición).
+ * Deshabilitado → 403 MODULO_PREMIUM_DESHABILITADO (con precio y tooltip para la UI). Si el codigo no existe, también se bloquea.
+ */
+function crearGuard({ repo = repoSql, ttlMs = 15_000, ahora = () => Date.now() } = {}) {
+  const cache = new Map();
+
+  async function estado(codigo) {
+    const c = cache.get(codigo);
+    if (c && ahora() - c.en < ttlMs) return c.valor;
+    const fila = await repo.leer(codigo);
+    const valor = fila ? { habilitado: Boolean(fila.habilitado), nombre: fila.nombre, precioUsd: Number(fila.precio_usd) } : { habilitado: false, nombre: codigo, precioUsd: null };
+    cache.set(codigo, { valor, en: ahora() });
+    return valor;
+  }
+
+  function requireModulo(codigo) {
+    if (!porCodigo(codigo)) throw new Error(`[premium] Módulo desconocido: ${codigo}`); // error de programación: falla al montar las rutas
+    return async (_req, _res, next) => {
+      try {
+        const m = await estado(codigo);
+        if (!m.habilitado) {
+          return next(new AppError(403, 'MODULO_PREMIUM_DESHABILITADO', TOOLTIP_BLOQUEADO, { modulo: codigo, nombre: m.nombre, precio_usd: m.precioUsd }));
+        }
+        return next();
+      } catch (err) {
+        return next(err);
+      }
+    };
+  }
+
+  const invalidar = () => cache.clear();
+  return { requireModulo, estado, invalidar, repo };
+}
+
+module.exports = { crearGuard, guard: crearGuard() };

+// Aplica database/migrations/*.sql en orden, una sola vez cada una, separando lotes por líneas "GO".
+// Uso: npm run migrate
+require('dotenv').config({ path: require('path').resolve(__dirname, '..', '.env') });
+const fs = require('node:fs');
+const path = require('node:path');
+const { getPool, closePool, sql } = require('../config/db');
+
+const DIR = path.resolve(__dirname, '..', 'database', 'migrations');
+
+function lotes(texto) {
+  return texto
+    .split(/^\s*GO\s*$/gim)
+    .map((l) => l.trim())
+    .filter(Boolean);
+}
+
+async function main() {
+  const pool = await getPool();
+  // La tabla de control se crea con el primer lote de 001; si no existe aún, se asume vacía.
+  const existeControl = (await pool.request().query(
+    "SELECT OBJECT_ID(N'dbo.schema_migrations', N'U') AS id")).recordset[0].id;
+  const aplicadas = new Set(
+    existeControl ? (await pool.request().query('SELECT nombre FROM dbo.schema_migrations')).recordset.map((r) => r.nombre) : [],
+  );
+
+  const archivos = fs.readdirSync(DIR).filter((f) => f.endsWith('.sql')).sort();
+  for (const archivo of archivos) {
+    if (aplicadas.has(archivo)) {
+      console.log(`= ${archivo} (ya aplicada)`);
+      continue;
+    }
+    console.log(`> Aplicando ${archivo}`);
+    for (const lote of lotes(fs.readFileSync(path.join(DIR, archivo), 'utf8'))) {
+      await pool.request().batch(lote);
+    }
+    await pool.request().input('n', sql.NVarChar(150), archivo)
+      .query('INSERT INTO dbo.schema_migrations (nombre) VALUES (@n)');
+  }

+'use strict';
+const axios = require('axios');
+const cfg = require('../config/env');
+const { sql, ejecutar } = require('../config/db');
+const { e } = require('../errors/AppError');
+
+const CACHE_MS = 5 * 60_000;
+
+/** Interpreta la respuesta de la fuente (DolarAPI Venezuela: { fuente, promedio, fechaActualizacion }). */
+function extraerTasa(payload) {
+  const tasa = Number(payload && payload.promedio);
+  if (!Number.isFinite(tasa) || tasa <= 0) throw new Error('La fuente BCV no devolvió una tasa válida.');
+  const fecha = payload.fechaActualizacion ? new Date(payload.fechaActualizacion) : null;
+  return {
+    tasa,
+    fuente: `dolarapi:${payload.fuente || 'oficial'}`,
+    fechaValor: fecha && !Number.isNaN(fecha.getTime()) ? fecha : null,
+  };
+}
+
+const repoSql = {
+  async guardar({ tasa, fuente, fechaValor }) {
+    const r = await ejecutar(
+      `INSERT INTO dbo.tasas_bcv (tasa, fuente, fecha_valor) OUTPUT inserted.capturada_en VALUES (@tasa, @fuente, @fecha)`,
+      { tasa: [sql.Decimal(18, 4), tasa], fuente: [sql.NVarChar(120), fuente], fecha: [sql.DateTime2, fechaValor] },
+    );
+    return r.recordset[0].capturada_en;
+  },
+  async ultima() {
+    const r = await ejecutar('SELECT TOP 1 tasa, fuente, fecha_valor, capturada_en FROM dbo.tasas_bcv ORDER BY capturada_en DESC');
+    return r.recordset[0] || null;
+  },
+};
+
+/**
+ * Servicio de tasa oficial BCV.
+ *  - `fresca: true` consulta la fuente en ese instante (alertas de WhatsApp: "al segundo exacto del envío").
+ *  - Sin `fresca`, reutiliza la captura de los últimos 5 min (formulario público).
+ *  - Si la fuente cae, usa la última tasa guardada SOLO si tiene <= BCV_MAX_ANTIGUEDAD_HORAS y la marca `desactualizada: true`.
+ *    Pasado ese límite responde 503 TASA_BCV_NO_DISPONIBLE: nunca se cobra con una tasa vieja sin avisar.
+ */
+function crearBcvService({ http = axios, repo = repoSql, url = cfg.bcv.url, timeoutMs = cfg.bcv.timeoutMs, maxHoras = cfg.bcv.maxAntiguedadHoras, ahora = () => Date.now() } = {}) {
+  let cache = null;
+
+  async function obtenerTasa({ fresca = false } = {}) {
+    if (!fresca && cache && ahora() - cache.en < CACHE_MS) return cache.valor;
+    try {
+      const { data } = await http.get(url, { timeout: timeoutMs, headers: { Accept: 'application/json' } });
+      const t = extraerTasa(data);
+      let capturadaEn = new Date(ahora());
+      try {
+        capturadaEn = await repo.guardar(t);
+      } catch (err) {
+        console.error('[bcv] No se pudo guardar la tasa (se usa igual):', err.message);
+      }
+      const valor = { tasa: t.tasa, fuente: t.fuente, fechaValor: t.fechaValor, capturadaEn, desactualizada: false };
+      cache = { valor, en: ahora() };
+      return valor;
+    } catch (err) {
+      console.error('[bcv] Fuente no disponible:', err.message);
+      let ultima = null;
+      try {
+        ultima = await repo.ultima();
+      } catch (dbErr) {
+        console.error('[bcv] Tampoco se pudo leer la última tasa guardada:', dbErr.message);
+      }
+      if (ultima) {
+        const horas = (ahora() - new Date(ultima.capturada_en).getTime()) / 3_600_000;
+        if (horas <= maxHoras) {

+  if (!raw || !cfg.drive.carpetaPadreId) {
+    throw e.noDisponible('DRIVE_NO_CONFIGURADO', 'La integración con Google Drive no está configurada (GOOGLE_SERVICE_ACCOUNT_JSON / GOOGLE_DRIVE_PARENT_FOLDER_ID).');
+  }
+  let credenciales;
+  try {
+    credenciales = JSON.parse(raw.trim().startsWith('{') ? raw : fs.readFileSync(raw, 'utf8'));
+  } catch {

+  }

+/**
+ * Google Drive: crea la carpeta 'Nombre_Apellido_ID' y sube los documentos DIRECTAMENTE desde memoria
+ * (Readable.from(buffer)), sin escribir nada en disco. Si algo falla, la carpeta parcial se envía a la papelera.
+ * IMPORTANTE: una cuenta de servicio no tiene cuota de almacenamiento propia; la carpeta padre debe estar en una
+ * Unidad Compartida (Shared Drive) o los archivos fallarán con "storageQuotaExceeded".
+ */
+function crearDriveService({ drive, carpetaPadreId = cfg.drive.carpetaPadreId } = {}) {
+  let cliente = drive || null;
+  const d = () => (cliente ||= clienteDriveReal());
+
+  async function enviarAPapelera(fileId) {
+    try {
+      await d().files.update({ fileId, requestBody: { trashed: true }, supportsAllDrives: true });
+    } catch (err) {
+      console.error(`[drive] No se pudo limpiar la carpeta ${fileId}:`, err.message);
+    }
+  }
+
+  /** archivos: [{ tipo:'FOTO_CARNET', buffer, mime, ext }] → { carpetaId, carpetaUrl, documentos:[{tipo, fileId, url, nombre}] } */
+  async function subirExpediente({ estudianteId, nombres, apellidos, archivos }) {
+    let carpetaId = null;
+    try {
+      const carpeta = await d().files.create({
+        requestBody: { name: nombreCarpeta(nombres, apellidos, estudianteId), mimeType: MIME_CARPETA, parents: [carpetaPadreId] },
+        fields: 'id, webViewLink',
+        supportsAllDrives: true,
+      });
+      carpetaId = carpeta.data.id;
+
+      const documentos = await Promise.all(archivos.map(async (a) => {
+        const nombre = `${a.tipo.toLowerCase()}.${a.ext}`;
+        const r = await d().files.create({
+          requestBody: { name: nombre, parents: [carpetaId] },
+          media: { mimeType: a.mime, body: Readable.from([a.buffer]) },
+          fields: 'id, webViewLink',
+          supportsAllDrives: true,
+        });
+        return { tipo: a.tipo, fileId: r.data.id, url: r.data.webViewLink || null, nombre, mime: a.mime, bytes: a.buffer.length };
+      }));
+
+      return { carpetaId, carpetaUrl: carpeta.data.webViewLink || null, documentos };
+    } catch (err) {
+      console.error('[drive] Falló la subida del expediente:', err.message);
+      if (carpetaId) await enviarAPapelera(carpetaId);
+      if (err && err.name === 'AppError') throw err;
+      throw e.externo('DRIVE_ERROR', 'No se pudo guardar la documentación en Google Drive. No se registró la inscripción; intenta de nuevo.');
+    }
+  }
+
+  return { subirExpediente, eliminarCarpeta: enviarAPapelera };

+/**
+ * Nómina mixta. Modalidades:
+ *   Fijo        → salario mensual fijo (las horas se registran pero no se pagan aparte)
+ *   Honorarios  → horas dictadas × tarifa por hora
+ *   Mixto       → salario fijo + horas × tarifa
+ * Aritmética exacta en centavos (utils/dinero).
+ */
+function calcularNomina({ modalidad, salarioFijoUsd = 0, tarifaHoraUsd = 0, horas = 0 }) {
+  if (!['Fijo', 'Honorarios', 'Mixto'].includes(modalidad)) throw new Error(`Modalidad desconocida: ${modalidad}`);
+  const paga = { fijo: modalidad !== 'Honorarios', horas: modalidad !== 'Fijo' };
+  const montoFijoUsd = paga.fijo ? Number(salarioFijoUsd) : 0;
+  const tarifa = paga.horas ? Number(tarifaHoraUsd) : 0;
+  const montoHorasUsd = paga.horas ? horasPorTarifa(horas, tarifa) : 0;
+  return {
+    montoFijoUsd,
+    horasTotales: Number(horas),
+    tarifaHoraUsd: tarifa,
+    montoHorasUsd,
+    totalUsd: sumar(montoFijoUsd, montoHorasUsd),
+  };

+function crearWhatsappService({ http = axios, token = cfg.whatsapp.token, phoneId = cfg.whatsapp.phoneNumberId, version = cfg.whatsapp.graphVersion } = {}) {

+'use strict';
+/**
+ * Entorno de pruebas SIN dependencias de red ni base de datos real.
+ *  - Sustituye los paquetes externos pesados (mssql, dotenv, axios, googleapis) por stubs inertes.
+ *  - Sustituye config/db por una base de datos falsa programable (`db.handler`) que registra cada consulta.
+ * Las pruebas verifican la lógica de negocio y la orquestación; el SQL real se valida en el test de consistencia de esquema
+ * y, contra un SQL Server real, con `npm run migrate` + smoke test (ver CAMBIOS_BMA_CORE.md).
+ */
+const Module = require('node:module');
+
+const sqlStub = new Proxy({}, {
+  get: (_t, k) => {
+    if (k === 'ISOLATION_LEVEL') return { READ_COMMITTED: 2 };
+    if (k === 'MAX') return 'MAX';
+    return (...a) => ({ tipo: String(k), a });
+  },
+});
+
+const externos = {
+  mssql: sqlStub,
+  dotenv: { config: () => ({}) },
+  axios: { get: async () => { throw new Error('axios real no disponible en pruebas'); }, post: async () => { throw new Error('axios real no disponible en pruebas'); } },
+  googleapis: { google: { auth: { GoogleAuth: class {} }, drive: () => ({}) } },
+  multer: Object.assign(() => ({ fields: () => () => {} }), { memoryStorage: () => ({}) }),
+};
+
+const db = {
+  sql: sqlStub,
+  llamadas: [],
+  rollbacks: 0,
+  commits: 0,
+  handler: async () => ({ recordset: [], rowsAffected: [0] }),
+  async ejecutar(texto, params = {}) {
+    db.llamadas.push({ texto, params });
+    const r = await db.handler(texto, params);
+    return r || { recordset: [], rowsAffected: [0] };
+  },
+  async conTransaccion(fn) {
+    try {
+      const r = await fn({ tx: true });
+      db.commits += 1;
+      return r;
+    } catch (err) {
+      db.rollbacks += 1;
+      throw err;
+    }
+  },
+  async getPool() { return {}; },
+  async closePool() {},
+  reiniciar() { db.llamadas.length = 0; db.rollbacks = 0; db.commits = 0; db.handler = async () => ({ recordset: [], rowsAffected: [0] }); },
+};
+
+const cargaOriginal = Module._load;
+Module._load = function cargar(request, ...resto) {
+  if (Object.prototype.hasOwnProperty.call(externos, request)) return externos[request];
+  if (/(^|[\\/])config[\\/]db$/.test(request)) return db;
+  return cargaOriginal.call(this, request, ...resto);
+};

+'use strict';
+
+/** Detecta el tipo real por "magic bytes": el mimetype que declara el navegador es manipulable. */
+function detectarTipo(buf) {
+  if (!Buffer.isBuffer(buf) || buf.length < 4) return null;
+  if (buf[0] === 0xff && buf[1] === 0xd8 && buf[2] === 0xff) return { mime: 'image/jpeg', ext: 'jpg' };
+  if (buf[0] === 0x89 && buf[1] === 0x50 && buf[2] === 0x4e && buf[3] === 0x47) return { mime: 'image/png', ext: 'png' };
+  if (buf[0] === 0x25 && buf[1] === 0x50 && buf[2] === 0x44 && buf[3] === 0x46) return { mime: 'application/pdf', ext: 'pdf' };
+  return null;
+}
+
+/** 'María José' → 'María_José'; elimina todo lo que no sea letra, dígito, guion o guion bajo (anti path-traversal en nombres de Drive). */
+function segmentoSeguro(texto) {
+  return String(texto).trim().replace(/\s+/g, '_').replace(/[^\p{L}\p{N}_-]/gu, '').slice(0, 40);
+}
+
+/** Nombre de carpeta en Drive: 'Nombre_Apellido_ID' (primer nombre + primer apellido + id del estudiante). */
+function nombreCarpeta(nombres, apellidos, id) {
+  const primero = (s) => segmentoSeguro(String(s).trim().split(/\s+/)[0] || 'X');
+  return `${primero(nombres)}_${primero(apellidos)}_${id}`;
+}

+/**
+ * Aritmética monetaria exacta con BigInt (centavos y tasa × 10⁴). Evita los errores de coma flotante:
+ *   0.1 + 0.2 !== 0.3, pero aquí todo se calcula en enteros y se redondea una sola vez (mitad hacia arriba).
+ */
+const aCentavos = (monto) => BigInt(Math.round(Number(monto) * 100));
+const deCentavos = (c) => Number(c) / 100;
+
+/** USD → Bs. con la tasa BCV (Bs. por USD). Devuelve un número con 2 decimales. */
+function usdABs(montoUsd, tasa) {
+  const t = BigInt(Math.round(Number(tasa) * 10000));
+  const bsCentavos = (aCentavos(montoUsd) * t + 5000n) / 10000n;
+  return deCentavos(bsCentavos);
+}
+
+function sumar(...montos) {
+  return deCentavos(montos.reduce((acc, m) => acc + aCentavos(m), 0n));
+}
+
+/** horas × tarifa por hora (ambos con 2 decimales) → USD con 2 decimales. */
+function horasPorTarifa(horas, tarifaUsd) {
+  const h = BigInt(Math.round(Number(horas) * 100));
+  const total = (h * aCentavos(tarifaUsd) + 50n) / 100n;
+  return deCentavos(total);
+}
+

+'use strict';

+/** Caracas = UTC-4 todo el año (sin horario de verano). Toda regla de "hoy" se evalúa en ese huso. */
+const OFFSET_CARACAS_MS = 4 * 3600_000;
+
+function hoyCaracas(ahora = Date.now()) {
+  return new Date(ahora - OFFSET_CARACAS_MS).toISOString().slice(0, 10);
+}
+
+const FECHA_RE = /^(\d{4})-(\d{2})-(\d{2})$/;
+
+function esFechaValida(s) {
+  const m = FECHA_RE.exec(String(s));
+  if (!m) return false;
+  const [y, mo, d] = [Number(m[1]), Number(m[2]), Number(m[3])];
+  const f = new Date(Date.UTC(y, mo - 1, d));
+  return f.getUTCFullYear() === y && f.getUTCMonth() === mo - 1 && f.getUTCDate() === d;
+}
+
+/** 'YYYY-MM-DD' → Date a medianoche UTC (lo que node-mssql espera para sql.Date sin desfase de huso). */
+function aFechaSql(s) {
+  return new Date(`${s}T00:00:00Z`);
+}
+
+/** Diferencia en días calendario b - a (ambas 'YYYY-MM-DD'). */
+function diasEntre(a, b) {
+  const t = (s) => { const [y, m, d] = s.split('-').map(Number); return Date.UTC(y, m - 1, d); };
+  return Math.round((t(b) - t(a)) / 86_400_000);
+}
+
+function periodoActual(ahora = Date.now()) {
+  return hoyCaracas(ahora).slice(0, 7);
+}

+const PERIODO_RE = /^\d{4}-(0[1-9]|1[0-2])$/;
+const esPeriodoValido = (p) => PERIODO_RE.test(String(p));
+
+/** 'YYYY-MM' → { desde:'YYYY-MM-01', hasta:'YYYY-MM-<último>', inicioUtc:Date, finUtc:Date } (límites a medianoche de Caracas). */
+function limitesPeriodo(periodo) {
+  const [y, m] = periodo.split('-').map(Number);
+  const ultimo = new Date(Date.UTC(y, m, 0)).getUTCDate();
+  const pad = (n) => String(n).padStart(2, '0');
+  return {
+    desde: `${periodo}-01`,
+    hasta: `${periodo}-${pad(ultimo)}`,
+    inicioUtc: new Date(Date.UTC(y, m - 1, 1) + OFFSET_CARACAS_MS),
+    finUtc: new Date(Date.UTC(y, m, 1) + OFFSET_CARACAS_MS),

+/** Semáforo no bloqueante: limita las cargas pesadas simultáneas (memoria de buffers + transacciones abiertas). */
+class Semaforo {
+  constructor(max) {
+    this.max = max;
+    this.enUso = 0;
+  }
+

+    if (this.enUso >= this.max) return null;
+    this.enUso += 1;

+      if (liberado) return;
+      liberado = true;
+      this.enUso -= 1;
+    };
+  }
+}

+'use strict';
+const { e } = require('../errors/AppError');
+const { esFechaValida, hoyCaracas } = require('./fechas');
+
+const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;
+
+/**
+ * Validador declarativo sin dependencias. `reglas[campo]` = { tipo, requerido, max, min, valores }.
+ * Tipos: texto | email | entero | decimal | fecha | fechaPasada | enum | booleano.
+ * Acumula TODOS los errores y lanza un único AppError 422 VALIDACION con `detalles: [{campo, mensaje}]`.
+ * Devuelve un objeto limpio SOLO con los campos declarados (descarta propiedades inesperadas).
+ */
+function validar(entrada, reglas) {
+  const src = entrada && typeof entrada === 'object' ? entrada : {};
+  const salida = {};
+  const errores = [];
+
+  for (const [campo, r] of Object.entries(reglas)) {
+    let v = src[campo];
+    if (typeof v === 'string') v = v.trim();
+    const vacio = v === undefined || v === null || v === '';
+    if (vacio) {
+      if (r.requerido) errores.push({ campo, mensaje: 'Es obligatorio.' });
+      else salida[campo] = r.defecto !== undefined ? r.defecto : null;
+      continue;
+    }
+    const falla = (mensaje) => errores.push({ campo, mensaje });
+
+    switch (r.tipo || 'texto') {
+      case 'texto':
+      case 'email': {
+        if (typeof v !== 'string') { falla('Debe ser texto.'); break; }
+        if (r.max && v.length > r.max) { falla(`Máximo ${r.max} caracteres.`); break; }
+        if (r.min && v.length < r.min) { falla(`Mínimo ${r.min} caracteres.`); break; }
+        if (r.tipo === 'email' && !EMAIL_RE.test(v)) { falla('Correo inválido.'); break; }
+        if (r.patron && !r.patron.test(v)) { falla(r.mensajePatron || 'Formato inválido.'); break; }
+        salida[campo] = r.tipo === 'email' ? v.toLowerCase() : v;
+        break;
+      }
+      case 'entero': {
+        const n = Number(v);
+        if (!Number.isInteger(n)) { falla('Debe ser un entero.'); break; }
+        if (r.min !== undefined && n < r.min) { falla(`Debe ser >= ${r.min}.`); break; }
+        if (r.max !== undefined && n > r.max) { falla(`Debe ser <= ${r.max}.`); break; }
+        salida[campo] = n;
+        break;
+      }
+      case 'decimal': {
+        const n = typeof v === 'number' ? v : Number(String(v).replace(',', '.'));
+        if (!Number.isFinite(n)) { falla('Debe ser un número.'); break; }
+        if (Math.round(n * 100) / 100 !== n) { falla('Máximo 2 decimales.'); break; }
+        if (r.min !== undefined && n < r.min) { falla(`Debe ser >= ${r.min}.`); break; }
+        if (r.max !== undefined && n > r.max) { falla(`Debe ser <= ${r.max}.`); break; }
+        if (r.positivo && n <= 0) { falla('Debe ser mayor que 0.'); break; }
+        salida[campo] = n;
+        break;
+      }
+      case 'fecha':
+      case 'fechaPasada': {
+        if (!esFechaValida(v)) { falla('Fecha inválida (AAAA-MM-DD).'); break; }
+        if (r.tipo === 'fechaPasada' && v > hoyCaracas()) { falla('No puede ser una fecha futura.'); break; }
+        if (r.min && v < r.min) { falla(`Debe ser >= ${r.min}.`); break; }
+        salida[campo] = v;
+        break;
+      }
+      case 'enum': {
+        if (!r.valores.includes(v)) { falla(`Valor no permitido. Opciones: ${r.valores.join(', ')}.`); break; }
+        salida[campo] = v;
+        break;
+      }
+      case 'booleano': {
+        if (v === true || v === 'true' || v === '1' || v === 1) salida[campo] = true;
+        else if (v === false || v === 'false' || v === '0' || v === 0) salida[campo] = false;
+        else falla('Debe ser verdadero o falso.');
+        break;
+      }
+      default:
+        throw new Error(`Tipo de regla desconocido: ${r.tipo}`);
+    }
+  }
+
+  if (errores.length) throw e.validacion(errores);
+  return salida;
+}
+
+/** Paginación segura: página >= 1, límite 1..200. */
+function paginacion(query, limiteMax = 200, limiteDefecto = 50) {
